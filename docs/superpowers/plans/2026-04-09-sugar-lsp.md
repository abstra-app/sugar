Create `abstra_sugar/script_parser.py`:

```python
import re
from typing import List, Tuple

from .ast import (
    ScriptCatch,
    ScriptClass,
    ScriptCollection,
    ScriptComment,
    ScriptElse,
    ScriptElseIf,
    ScriptFinally,
    ScriptForLoop,
    ScriptFunction,
    ScriptIf,
    ScriptNode,
    ScriptStatement,
    ScriptTemplateLiteral,
    ScriptTry,
    ScriptWhileLoop,
)


def parse_script(body: str) -> List[ScriptNode]:
    parsed = _parse_lines(body)
    nodes: List[ScriptNode] = []
    i = 0
    while i < len(parsed):
        node, i = _parse_script_node(parsed, i, in_class=False)
        if node is not None:
            nodes.append(node)
    return nodes


def _parse_lines(body: str) -> List[Tuple[int, str, bool]]:
    parsed: List[Tuple[int, str, bool]] = []
    for line in body.split("\n"):
        if not line.strip():
            parsed.append((0, "", True))
            continue
        indent = len(line) - len(line.lstrip(" "))
        parsed.append((indent, line.strip(), False))
    return parsed


def _collect_block(
    parsed: List[Tuple[int, str, bool]], start: int, parent_indent: int
) -> Tuple[List[Tuple[int, str, bool]], int]:
    block: List[Tuple[int, str, bool]] = []
    i = start
    while i < len(parsed):
        indent, content, is_blank = parsed[i]
        if is_blank:
            block.append(parsed[i])
            i += 1
            continue
        if indent <= parent_indent:
            break
        block.append(parsed[i])
        i += 1
    return block, i


def _parse_block(
    parsed: List[Tuple[int, str, bool]], start: int, parent_indent: int,
    in_class: bool = False,
) -> Tuple[List[ScriptNode], int]:
    block_lines, end = _collect_block(parsed, start, parent_indent)
    nodes: List[ScriptNode] = []
    j = 0
    while j < len(block_lines):
        node, j = _parse_script_node(block_lines, j, in_class=in_class)
        if node is not None:
            nodes.append(node)
    return nodes, end


def _parse_script_node(
    parsed: List[Tuple[int, str, bool]], i: int, in_class: bool = False,
) -> Tuple[ScriptNode | None, int]:
    if i >= len(parsed):
        return None, i

    indent, content, is_blank = parsed[i]
    if is_blank:
        return None, i + 1

    # Comment
    if content.startswith("//"):
        text = content[2:].lstrip()
        return ScriptComment(text=text), i + 1

    # Block constructs (end with :)
    if content.endswith(":"):
        header = content[:-1].rstrip()
        return _parse_block_construct(parsed, i, indent, header, in_class)

    # Plain statement
    return ScriptStatement(text=content), i + 1


def _parse_block_construct(
    parsed: List[Tuple[int, str, bool]], i: int, indent: int,
    header: str, in_class: bool,
) -> Tuple[ScriptNode, int]:

    # class Name: / class Name extends Base:
    m = re.match(r"class\s+(\w+)(?:\s+extends\s+(\w+))?$", header)
    if m:
        name, extends = m.group(1), m.group(2)
        methods, end = _parse_block(parsed, i + 1, indent, in_class=True)
        return ScriptClass(name=name, extends=extends, methods=methods), end

    # async function: async name(args):
    m = re.match(r"async\s+(\w+)\s*\(([^)]*)\)$", header)
    if m:
        name, args_str = m.group(1), m.group(2)
        params = [p.strip() for p in args_str.split(",") if p.strip()]
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptFunction(name=name, params=params, is_async=True, body=body), end

    # function/method: name(args):
    m = re.match(r"(\w+)\s*\(([^)]*)\)$", header)
    if m:
        name, args_str = m.group(1), m.group(2)
        params = [p.strip() for p in args_str.split(",") if p.strip()]
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptFunction(name=name, params=params, is_async=False, body=body), end

    # for var of/in iterable:
    m = re.match(r"for\s+(.+?)\s+(of|in)\s+(.+)", header)
    if m:
        var, keyword, iterable = m.groups()
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptForLoop(var=var, keyword=keyword, iterable=iterable, body=body), end

    # while condition:
    m = re.match(r"while\s+(.+)", header)
    if m:
        condition = m.group(1)
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptWhileLoop(condition=condition, body=body), end

    # if condition:
    m = re.match(r"if\s+(.+)", header)
    if m:
        condition = m.group(1)
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptIf(condition=condition, body=body), end

    # else if condition:
    m = re.match(r"else\s+if\s+(.+)", header)
    if m:
        condition = m.group(1)
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptElseIf(condition=condition, body=body), end

    # else:
    if header == "else":
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptElse(body=body), end

    # try:
    if header == "try":
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptTry(body=body), end

    # catch(e):
    m = re.match(r"catch\s*\(([^)]*)\)$", header)
    if m:
        param = m.group(1).strip() or None
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptCatch(param=param, body=body), end

    # finally:
    if header == "finally":
        body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
        return ScriptFinally(body=body), end

    # collection: lhs =:
    m = re.match(r"(.+?)\s*=\s*$", header)
    if m:
        lhs = m.group(1)
        value, end = _compile_collection(parsed, i + 1, indent)
        return ScriptCollection(lhs=lhs, value=value), end

    # nested object: lhs = key:
    m = re.match(r"(.+?)\s*=\s*(.+)", header)
    if m:
        lhs = m.group(1)
        first_key = m.group(2)
        value, end = _compile_collection(parsed, i + 1, indent)
        return ScriptCollection(lhs=lhs, value="{" + first_key + ": " + value.strip("{}") + "}"), end

    # template literal: prefix html!: / text!: / table!: / htm!:
    tmpl_m = re.match(r"(.*?)(html!|htm!|css!|js!|text!|table!)\s*$", header)
    if tmpl_m:
        prefix = tmpl_m.group(1)
        literal_type = tmpl_m.group(2)
        content, end = _compile_template(parsed, i + 1, indent, literal_type)
        return ScriptTemplateLiteral(prefix=prefix, literal_type=literal_type, content=content), end

    # trailing arrow: expr ending in (...):
    # e.g. document.addEventListener("click", (e):
    if header.endswith(")"):
        depth = 0
        for k in range(len(header) - 1, -1, -1):
            if header[k] == ")":
                depth += 1
            elif header[k] == "(":
                depth -= 1
                if depth == 0:
                    # This is a block with trailing arrow
                    body, end = _parse_block(parsed, i + 1, indent, in_class=in_class)
                    return ScriptFunction(
                        name="",
                        params=[p.strip() for p in header[k+1:-1].split(",") if p.strip()],
                        is_async=False,
                        body=body,
                    ), end

    # Fallback: unknown block construct — treat header + block as statement
    _, end = _collect_block(parsed, i + 1, indent)
    return ScriptStatement(text=header + ":"), end


def _compile_collection(
    parsed: List[Tuple[int, str, bool]], start: int, parent_indent: int,
) -> Tuple[str, int]:
    """Compile indented block into {object} or [array] string."""
    pairs: List[str] = []
    has_keys = False
    has_item_separators = False
    i = start

    while i < len(parsed):
        indent, content, is_blank = parsed[i]
        if is_blank:
            i += 1
            continue
        if indent <= parent_indent:
            break

        # standalone ":" → object item separator in array
        if content == ":":
            has_item_separators = True
            obj, i = _compile_collection(parsed, i + 1, indent)
            pairs.append(obj)
            continue

        if content.endswith(":") and not content.endswith("::"):
            key = content[:-1].rstrip()
            # check if next lines are children
            if i + 1 < len(parsed) and not parsed[i + 1][2] and parsed[i + 1][0] > indent:
                has_keys = True
                sub, i = _compile_collection(parsed, i + 1, indent)
                pairs.append(f"{key}: {sub}")
            else:
                has_keys = True
                pairs.append(content[:-1].rstrip() + ": ")
                i += 1
        else:
            if re.match(r'[\w"\']+\s*:', content) and not content.startswith("//"):
                has_keys = True
            pairs.append(content)
            i += 1

    if has_item_separators:
        return "[" + ", ".join(pairs) + "]", i
    if has_keys:
        return "{" + ", ".join(pairs) + "}", i
    return "[" + ", ".join(pairs) + "]", i


def _compile_template(
    parsed: List[Tuple[int, str, bool]], start: int, parent_indent: int,
    literal_type: str,
) -> Tuple[str, int]:
    """Collect template block lines and compile them."""
    block_lines, end = _collect_block(parsed, start, parent_indent)
    if not block_lines:
        return "", end

    base_indent = parent_indent + 1
    src_lines: List[str] = []
    for ind, content, is_blank in block_lines:
        if is_blank:
            continue
        src_lines.append(" " * (ind - base_indent) + content)
    src = "\n".join(src_lines)

    if literal_type in ("html!", "htm!"):
        from .lexer import scan
        from .parser import parse
        from .compiler import compile as compile_nodes

        nodes = parse(scan(src))
        html = compile_nodes(nodes, data=None).rstrip("\n")
        html = re.sub(r"(?<!\$)\{([^}]+)\}", r"${\1}", html)
        if literal_type == "htm!":
            html = re.sub(r'="(\$\{[^}]+\})"', r"=\1", html)
        html = re.sub(r"\n\t*", "", html)
        return html, end

    if literal_type == "text!":
        text = "\\n".join(src_lines)
        text = re.sub(r"\{([^}]+)\}", r"${\1}", text)
        return text, end

    if literal_type == "table!":
        return _compile_table_literal_from_lines(src_lines), end

    # css!/js! — pass through
    return src.replace("\n", "\\n"), end


def _compile_table_literal_from_lines(src_lines: List[str]) -> str:
    if not src_lines:
        return "[]"
    headers = [h.strip() for h in src_lines[0].split("|")]
    rows: List[str] = []
    for line in src_lines[1:]:
        values = [v.strip() for v in line.split("|")]
        pairs = []
        for key, val in zip(headers, values):
            if key and val:
                pairs.append(f"{key}: {val}")
        rows.append("{" + ", ".join(pairs) + "}")
    return "[" + ", ".join(rows) + "]"
```

from typing import Dict, List

from lsprotocol import types

from ..ast import ComponentDef, Node
from ..lexer import scan
from ..parser import parse
from ..tokens import Token


class DocumentState:
    def __init__(self, uri: str, source: str):
        self.uri = uri
        self.source = source
        self.tokens: List[Token] = []
        self.ast: List[Node] = []
        self.components: Dict[str, ComponentDef] = {}
        self.diagnostics: List[types.Diagnostic] = []
        self._parse()

    def update(self, source: str) -> None:
        self.source = source
        self._parse()

    def _parse(self) -> None:
        self.diagnostics = []
        try:
            self.tokens = scan(self.source)
            self.ast = parse(self.tokens)
            self.components = self._collect_components(self.ast)
        except Exception as e:
            self.ast = []
            self.components = {}
            self.diagnostics.append(
                types.Diagnostic(
                    range=types.Range(
                        start=types.Position(line=0, character=0),
                        end=types.Position(line=0, character=0),
                    ),
                    message=str(e),
                    severity=types.DiagnosticSeverity.Error,
                    source="sugar",
                )
            )

    def _collect_components(self, nodes: List[Node]) -> Dict[str, ComponentDef]:
        components: Dict[str, ComponentDef] = {}
        for node in nodes:
            if isinstance(node, ComponentDef):
                components[node.name] = node
        return components
```

        d_line = line - prev_line
        d_col = col - prev_col if d_line == 0 else col
        result.extend([d_line, d_col, length, type_idx, mod_mask])
        prev_line = line
        prev_col = col
    return result


def _collect_tokens(source: str) -> List[RawToken]:
    """Parse source and walk AST to collect all semantic tokens."""
    raw_tokens = scan(source)
    ast = parse(raw_tokens)

    tokens: List[RawToken] = []
    lines = source.split("\n")

    # Walk raw tokens to get line positions, then match with AST
    line_num = 0
    for raw_token in raw_tokens:
        if raw_token.type == "blank":
            line_num += 1
            continue

        line_text = lines[line_num] if line_num < len(lines) else ""
        indent = raw_token.indent

        if raw_token.type == "comment":
            tokens.append((line_num, indent, len(line_text.strip()), _type_idx("comment"), 0))
            line_num += 1
            continue

        if raw_token.type != "line":
            line_num += 1
            continue

        _tokenize_line(tokens, raw_token, line_num, line_text, ast)
        line_num += 1

    return tokens


def _tokenize_line(
    tokens: List[RawToken], raw_token: Token, line_num: int,
    line_text: str, ast: List[Node],
) -> None:
    """Emit semantic tokens for a single line based on its token and context."""
    head = raw_token.head
    text = raw_token.text
    indent = raw_token.indent
    stripped = line_text.strip()

    # Determine context from AST
    context = _get_context(raw_token, ast)

    if context == "script_body":
        # Script body lines are handled separately (raw body)
        _tokenize_script_line(tokens, line_num, line_text, indent)
        return

    if context == "style":
        _tokenize_style_line(tokens, line_num, line_text, indent, head, text, raw_token.has_colon)
        return

    # HTML context
    _tokenize_html_line(tokens, line_num, line_text, indent, head, text, raw_token.has_colon)


def _get_context(raw_token: Token, ast: List[Node]) -> str:
    """Determine if this token is in script_body, style, or html context."""
    if raw_token.type in ("script_body", "markdown_body", "math_body", "svg_body"):
        return "script_body"
    return "html"


def _tokenize_html_line(
    tokens: List[RawToken], line_num: int, line_text: str,
    indent: int, head: str, text: str, has_colon: bool,
) -> None:
    """Emit tokens for an HTML-context line."""
    stripped = line_text.strip()
    pos = indent  # current position in line_text

    # Component def: name = (params):
    m = re.match(r"(\w+)\s*=\s*\(([^)]*)\)$", head)
    if m:
        name_start = line_text.index(m.group(1), indent)
        tokens.append((line_num, name_start, len(m.group(1)), _type_idx("function"), _mod_mask("declaration")))
        if m.group(2):
            params_str = m.group(2)
            params_start = line_text.index("(", name_start) + 1
            for param in params_str.split(","):
                param = param.strip()
                if param:
                    p_start = line_text.index(param, params_start)
                    tokens.append((line_num, p_start, len(param), _type_idx("parameter"), 0))
                    params_start = p_start + len(param)
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Component call: name(args)
    m = re.match(r"(\w+)\s*\(([^)]*)\)$", head)
    if m and not _is_html_tag(m.group(1)):
        name_start = line_text.index(m.group(1), indent)
        tokens.append((line_num, name_start, len(m.group(1)), _type_idx("function"), 0))
        if m.group(2):
            args_start = line_text.index("(", name_start) + 1
            args_end = line_text.index(")", args_start)
            tokens.append((line_num, args_start, args_end - args_start, _type_idx("string"), 0))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # for var of/in iterable:
    m = re.match(r"for\s+(.+?)\s+(of|in)\s+(.+)", head)
    if m:
        kw_start = indent
        tokens.append((line_num, kw_start, 3, _type_idx("keyword"), 0))  # for
        var = m.group(1)
        var_start = line_text.index(var, kw_start + 3)
        tokens.append((line_num, var_start, len(var), _type_idx("variable"), 0))
        kw2 = m.group(2)
        kw2_start = line_text.index(f" {kw2} ", var_start) + 1
        tokens.append((line_num, kw2_start, len(kw2), _type_idx("keyword"), 0))
        iterable = m.group(3)
        it_start = kw2_start + len(kw2) + 1
        tokens.append((line_num, it_start, len(iterable), _type_idx("variable"), 0))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # if condition:
    if head.startswith("if "):
        tokens.append((line_num, indent, 2, _type_idx("keyword"), 0))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # slot:
    if head == "slot":
        tokens.append((line_num, indent, 4, _type_idx("keyword"), 0))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Implicit child: : text
    if not head and has_colon:
        tokens.append((line_num, indent, 1, _type_idx("operator"), 0))
        if text:
            _tokenize_interpolations(tokens, line_num, line_text, indent + 2, text)
        return

    # Literal macros: table!, markdown!, math!, svg!
    for macro in ("table!", "markdown!", "math!", "svg!"):
        if head == macro or head.startswith(macro + ".") or head.startswith(macro + "#"):
            macro_start = indent
            tokens.append((line_num, macro_start, len(macro), _type_idx("macro"), 0))
            rest = head[len(macro):]
            if rest:
                _tokenize_classes_and_ids(tokens, line_num, line_text, indent + len(macro), rest)
            if has_colon:
                colon_pos = line_text.rindex(":")
                tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
            return

    # Element: tag.class1.class2#id attr=value:
    _tokenize_element_head(tokens, line_num, line_text, indent, head, has_colon)

    # Text content with interpolations
    if text:
        text_start = line_text.index(text, indent)
        _tokenize_interpolations(tokens, line_num, line_text, text_start, text)


_HTML_TAGS = {

def _is_html_tag(name: str) -> bool:
    return name in _HTML_TAGS


def _tokenize_element_head(
    tokens: List[RawToken], line_num: int, line_text: str,
    indent: int, head: str, has_colon: bool,
) -> None:
    """Parse tag.class1.class2#id attr=value and emit tokens."""
    parts = head.split()
    if not parts:
        return

    tag_part = parts[0]
    pos = indent

    # Extract tag name (before . or #)
    tag_name = tag_part.split(".")[0].split("#")[0]

    if tag_name and _is_html_tag(tag_name):
        tag_start = line_text.index(tag_name, pos)
        tokens.append((line_num, tag_start, len(tag_name), _type_idx("tag"), 0))
        pos = tag_start + len(tag_name)

    # Classes and IDs in the tag part
    rest = tag_part[len(tag_name):] if tag_name else tag_part
    if rest:
        _tokenize_classes_and_ids(tokens, line_num, line_text, pos, rest)

    # Attributes: key=value pairs
    for attr_part in parts[1:]:
        if "=" in attr_part:
            key, val = attr_part.split("=", 1)
            key_start = line_text.index(key + "=", pos)
            tokens.append((line_num, key_start, len(key), _type_idx("property"), 0))
            val_start = key_start + len(key) + 1
            tokens.append((line_num, val_start, len(val), _type_idx("string"), 0))
            pos = val_start + len(val)
        else:
            # Boolean attribute
            attr_start = line_text.index(attr_part, pos)
            tokens.append((line_num, attr_start, len(attr_part), _type_idx("property"), 0))
            pos = attr_start + len(attr_part)

    if has_colon:
        colon_pos = line_text.rindex(":")
        tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))


def _tokenize_classes_and_ids(
    tokens: List[RawToken], line_num: int, line_text: str,
    start_pos: int, fragment: str,
) -> None:
    """Tokenize .class1.class2#id fragments."""
    pos = start_pos
    i = 0
    while i < len(fragment):
        if fragment[i] == ".":
            # Find end of class name
            j = i + 1
            while j < len(fragment) and fragment[j] not in (".", "#"):
                j += 1
            class_name = fragment[i:j]
            tokens.append((line_num, pos, len(class_name), _type_idx("class"), 0))
            pos += len(class_name)
            i = j
        elif fragment[i] == "#":
            j = i + 1
            while j < len(fragment) and fragment[j] not in (".", "#"):
                j += 1
            id_name = fragment[i:j]
            tokens.append((line_num, pos, len(id_name), _type_idx("type"), _mod_mask("id")))
            pos += len(id_name)
            i = j
        else:
            pos += 1
            i += 1


def _tokenize_interpolations(
    tokens: List[RawToken], line_num: int, line_text: str,
    start: int, text: str,
) -> None:
    """Find {expr} interpolations in text."""
    for m in re.finditer(r"\{([^}]+)\}", text):
        interp_start = start + m.start()
        tokens.append((line_num, interp_start, m.end() - m.start(), _type_idx("variable"), 0))


def _tokenize_style_line(
    tokens: List[RawToken], line_num: int, line_text: str,
    indent: int, head: str, text: str, has_colon: bool,
) -> None:
    """Emit tokens for a CSS-context line."""
    stripped = line_text.strip()

    # @-rule
    if stripped.startswith("@"):
        m = re.match(r"@(\w+)", stripped)
        if m:
            at_start = indent
            tokens.append((line_num, at_start, len(m.group(0)), _type_idx("keyword"), 0))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Mixin def: name = ():
    m = re.match(r"(\w+)\s*=\s*\(\)", head)
    if m:
        tokens.append((line_num, indent, len(m.group(1)), _type_idx("function"), _mod_mask("declaration")))
        if has_colon:
            colon_pos = line_text.rindex(":")
            tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Mixin call: @name()
    m = re.match(r"@(\w+)\(\)", head)
    if m:
        tokens.append((line_num, indent, len(stripped), _type_idx("function"), 0))
        return

    if has_colon and text:
        # property: value
        tokens.append((line_num, indent, len(head), _type_idx("property"), 0))
        val_start = line_text.index(text, indent + len(head))
        tokens.append((line_num, val_start, len(text), _type_idx("string"), 0))
    elif has_colon and not text:
        # selector:
        tokens.append((line_num, indent, len(head), _type_idx("class"), 0))
        colon_pos = line_text.rindex(":")
        tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))


def _tokenize_script_line(
    tokens: List[RawToken], line_num: int, line_text: str, indent: int,
) -> None:
    """Emit tokens for a JS-context line.

    Since the script body is stored as raw text in the token stream,
    we use regex-based matching on individual lines within the script body.
    The script AST (body_nodes) provides structural context, but for
    semantic tokens we need character-level positions from the source lines.
    """
    stripped = line_text.strip()
    if not stripped:
        return

    # Comment
    if stripped.startswith("//"):
        tokens.append((line_num, indent, len(stripped), _type_idx("comment"), 0))
        return

    # Keywords at start of line
    _JS_BLOCK_KW = {"for", "if", "else", "while", "class", "try", "catch", "finally"}
    _JS_STMT_KW = {"return", "throw", "await", "new", "const", "let", "var", "async", "break", "continue"}

    first_word = stripped.split()[0] if stripped.split() else ""

    # Block keywords
    if first_word in _JS_BLOCK_KW:
        tokens.append((line_num, indent, len(first_word), _type_idx("keyword"), 0))

    # Statement keywords
    if first_word in _JS_STMT_KW:
        tokens.append((line_num, indent, len(first_word), _type_idx("keyword"), 0))

    # of/in/extends as secondary keywords
    for kw in ("of", "in", "extends"):
        pattern = rf"\s{kw}\s"
        for m in re.finditer(pattern, line_text):
            tokens.append((line_num, m.start() + 1, len(kw), _type_idx("keyword"), 0))

    # Function/method def: name(args):
    m = re.match(r"(\s*)(async\s+)?(\w+)\s*\([^)]*\)\s*:$", line_text)
    if m:
        if m.group(2):
            tokens.append((line_num, len(m.group(1)), 5, _type_idx("keyword"), 0))
        name = m.group(3)
        name_start = line_text.index(name, len(m.group(1)) + (len(m.group(2)) if m.group(2) else 0))
        tokens.append((line_num, name_start, len(name), _type_idx("function"), _mod_mask("declaration")))
        # params
        paren_start = line_text.index("(", name_start)
        paren_end = line_text.index(")", paren_start)
        params_str = line_text[paren_start+1:paren_end]
        if params_str.strip():
            p_pos = paren_start + 1
            for param in params_str.split(","):
                param = param.strip()
                if param:
                    p_start = line_text.index(param, p_pos)
                    tokens.append((line_num, p_start, len(param), _type_idx("parameter"), 0))
                    p_pos = p_start + len(param)
        colon_pos = line_text.rindex(":")
        tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Class def: class Name: / class Name extends Base:
    m = re.match(r"(\s*)class\s+(\w+)(?:\s+extends\s+(\w+))?\s*:$", line_text)
    if m:
        name = m.group(2)
        name_start = line_text.index(name, len(m.group(1)) + 5)
        tokens.append((line_num, name_start, len(name), _type_idx("class"), _mod_mask("declaration")))
        if m.group(3):
            ext = m.group(3)
            ext_start = line_text.index(ext, name_start + len(name))
            tokens.append((line_num, ext_start, len(ext), _type_idx("class"), 0))
        colon_pos = line_text.rindex(":")
        tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))
        return

    # Template literal markers: html!, htm!, text!, table!
    for macro in ("html!", "htm!", "text!", "table!", "css!", "js!"):
        if macro in stripped:
            m_pos = line_text.index(macro)
            tokens.append((line_num, m_pos, len(macro), _type_idx("macro"), 0))

    # Collection: name =:
    if stripped.endswith("=:"):
        colon_pos = line_text.rindex(":")
        tokens.append((line_num, colon_pos - 1, 1, _type_idx("operator"), 0))
        tokens.append((line_num, colon_pos, 1, _type_idx("operator"), 0))

    # Strings
    for m in re.finditer(r'"[^"]*"|\'[^\']*\'', line_text):
        tokens.append((line_num, m.start(), m.end() - m.start(), _type_idx("string"), 0))

    # Numbers (not part of identifiers)
    for m in re.finditer(r'(?<![a-zA-Z_])\b\d+(\.\d+)?\b', line_text):
        tokens.append((line_num, m.start(), m.end() - m.start(), _type_idx("number"), 0))

    # Template literal ${expr}
    for m in re.finditer(r'\$\{[^}]+\}', line_text):
        tokens.append((line_num, m.start(), m.end() - m.start(), _type_idx("variable"), 0))

    # Interpolation {expr} (Sugar-style in text)
    for m in re.finditer(r'(?<!\$)\{([^}]+)\}', line_text):
        # Only outside strings
        tokens.append((line_num, m.start(), m.end() - m.start(), _type_idx("variable"), 0))

    # Built-in objects
    _BUILTINS = {"console", "document", "window", "Math", "JSON", "Object", "Array",
                 "String", "Number", "Boolean", "Date", "Promise", "Error"}
    for builtin in _BUILTINS:
        for m in re.finditer(rf'\b{builtin}\b', line_text):
            tokens.append((line_num, m.start(), len(builtin), _type_idx("variable"), _mod_mask("defaultLibrary")))

    # Constants
    for const in ("true", "false", "null", "undefined"):
        for m in re.finditer(rf'\b{const}\b', line_text):
            tokens.append((line_num, m.start(), len(const), _type_idx("keyword"), _mod_mask("defaultLibrary")))
```

from pygls.lsp.server import LanguageServer
from lsprotocol import types

from .semantic_tokens import TOKEN_MODIFIERS, TOKEN_TYPES, get_semantic_tokens
from .state import DocumentState


def create_server() -> LanguageServer:
    server = LanguageServer("sugar-lsp", "v0.1.0")
    documents: dict[str, DocumentState] = {}

    @server.feature(types.TEXT_DOCUMENT_DID_OPEN)
    def did_open(params: types.DidOpenTextDocumentParams):
        uri = params.text_document.uri
        documents[uri] = DocumentState(uri, params.text_document.text)
        _publish_diagnostics(server, documents[uri])

    @server.feature(types.TEXT_DOCUMENT_DID_CHANGE)
    def did_change(params: types.DidChangeTextDocumentParams):
        uri = params.text_document.uri
        doc = server.workspace.get_text_document(uri)
        if uri in documents:
            documents[uri].update(doc.source)
        else:
            documents[uri] = DocumentState(uri, doc.source)
        _publish_diagnostics(server, documents[uri])

    @server.feature(types.TEXT_DOCUMENT_DID_CLOSE)
    def did_close(params: types.DidCloseTextDocumentParams):
        documents.pop(params.text_document.uri, None)

    @server.feature(
        types.TEXT_DOCUMENT_SEMANTIC_TOKENS_FULL,
        types.SemanticTokensOptions(
            legend=types.SemanticTokensLegend(
                token_types=TOKEN_TYPES,
                token_modifiers=TOKEN_MODIFIERS,
            ),
            full=True,
        ),
    )
    def semantic_tokens_full(params: types.SemanticTokensParams):
        uri = params.text_document.uri
        doc = documents.get(uri)
        if doc is None:
            return types.SemanticTokens(data=[])
        data = get_semantic_tokens(doc.source)
        return types.SemanticTokens(data=data)

    return server


def _publish_diagnostics(server: LanguageServer, state: DocumentState) -> None:
    server.text_document_publish_diagnostics(
        types.PublishDiagnosticsParams(

from typing import Dict, List

from lsprotocol import types

from ..ast import ComponentCall, ComponentDef, Element, Node
from ..lexer import scan
from ..parser import HTML_TAGS, parse


def compute_diagnostics(source: str) -> List[types.Diagnostic]:
    diagnostics: List[types.Diagnostic] = []

    try:
        tokens = scan(source)
        ast = parse(tokens)
    except Exception as e:
        diagnostics.append(
            types.Diagnostic(
                range=types.Range(
                    start=types.Position(line=0, character=0),
                    end=types.Position(line=0, character=0),
                ),
                message=str(e),
                severity=types.DiagnosticSeverity.Error,
                source="sugar",
            )
        )
        return diagnostics

    # Collect component definitions
    components: Dict[str, ComponentDef] = {}
    _collect_components(ast, components)

    # Check for undefined component calls
    _check_undefined_calls(ast, components, diagnostics, source)

    return diagnostics


def _collect_components(nodes: List[Node], components: Dict[str, ComponentDef]) -> None:
    for node in nodes:
        if isinstance(node, ComponentDef):
            components[node.name] = node
        elif isinstance(node, Element):
            _collect_components(node.children, components)


def _check_undefined_calls(
    nodes: List[Node],
    components: Dict[str, ComponentDef],
    diagnostics: List[types.Diagnostic],
    source: str,
) -> None:
    lines = source.split("\n")
    for node in nodes:
        if isinstance(node, ComponentCall):
            if node.name not in components:
                # Find the line containing this call
                line_num = _find_line(lines, node.name + "(")
                diagnostics.append(
                    types.Diagnostic(
                        range=types.Range(
                            start=types.Position(line=line_num, character=0),
                            end=types.Position(line=line_num, character=len(lines[line_num]) if line_num < len(lines) else 0),
                        ),
                        message=f"Undefined component '{node.name}'",
                        severity=types.DiagnosticSeverity.Warning,
                        source="sugar",
                    )
                )
        elif isinstance(node, Element):
            _check_undefined_calls(node.children, components, diagnostics, source)
        elif isinstance(node, ComponentDef):
            _check_undefined_calls(node.children, components, diagnostics, source)


def _find_line(lines: List[str], needle: str) -> int:
    for i, line in enumerate(lines):
        if needle in line:
            return i
    return 0
```

    "position", "top", "right", "bottom", "left",

]

             "date", "time", "checkbox", "radio", "file", "submit",
             "button", "hidden", "range", "color", "search"],
    "method": ["get", "post", "put", "delete", "patch"],
    "target": ["_blank", "_self", "_parent", "_top"],
    "rel": ["stylesheet", "icon", "preconnect", "noopener", "noreferrer"],
    "charset": ["UTF-8"],

def get_completions(
    state: DocumentState, line: int, character: int,
) -> List[types.CompletionItem]:
    lines = state.source.split("\n")
    if line >= len(lines):
        return []

    current_line = lines[line]
    prefix = current_line[:character].strip()
    context = _get_line_context(lines, line)

    items: List[types.CompletionItem] = []

    if context == "style":
        # CSS property completions
        for prop in _CSS_PROPERTIES:
            if prop.startswith(prefix):
                items.append(types.CompletionItem(
                    label=prop,
                    kind=types.CompletionItemKind.Property,
                ))
        return items

    if context == "script":
        # JS keyword completions
        for kw in _JS_KEYWORDS:
            if kw.startswith(prefix):
                items.append(types.CompletionItem(
                    label=kw,
                    kind=types.CompletionItemKind.Keyword,
                ))
        return items

    # Check for attribute value context: "attr="
    attr_match = prefix.rsplit("=", 1)
    if len(attr_match) == 2 and " " not in attr_match[0].split()[-1]:
        attr_name = attr_match[0].split()[-1]
        if attr_name in _ATTR_VALUES:
            for val in _ATTR_VALUES[attr_name]:
                items.append(types.CompletionItem(
                    label=val,
                    kind=types.CompletionItemKind.Value,
                ))
            return items

    # HTML context: tags, keywords, components
    for tag in sorted(HTML_TAGS):
        if tag.startswith(prefix):
            items.append(types.CompletionItem(
                label=tag,
                kind=types.CompletionItemKind.Class,
            ))

    for kw in _KEYWORDS:
        if kw.startswith(prefix):
            items.append(types.CompletionItem(
                label=kw,
                kind=types.CompletionItemKind.Keyword,
            ))

    for name in state.components:
        if name.startswith(prefix):
            comp = state.components[name]
            params = ", ".join(comp.params)
            items.append(types.CompletionItem(
                label=name,
                kind=types.CompletionItemKind.Function,
                detail=f"{name}({params})",
            ))

    return items


def _get_line_context(lines: List[str], target_line: int) -> str:
    """Determine if target_line is inside a style: or script: block."""
    indent_at_target = len(lines[target_line]) - len(lines[target_line].lstrip()) if lines[target_line].strip() else 0

    for i in range(target_line - 1, -1, -1):
        line = lines[i]
        stripped = line.strip()
        if not stripped:
            continue
        line_indent = len(line) - len(line.lstrip())
        if line_indent < indent_at_target:
            tag = stripped.split(".")[0].split("#")[0].split()[0]
            if tag == "style" or (tag == "style:" or stripped.startswith("style")):
                if ":" in stripped:
                    return "style"
            if tag == "script" or stripped.startswith("script"):
                if ":" in stripped:
                    return "script"
            break
    return "html"
```

from lsprotocol import types

from .state import DocumentState


def get_definition(
    state: DocumentState, line: int, character: int,
) -> Optional[types.Location]:
    lines = state.source.split("\n")
    if line >= len(lines):
        return None

    current_line = lines[line]

    # Check if cursor is on a component call: name(...)
    m = re.match(r"(\s*)(\w+)\s*\(", current_line)
    if m:
        name = m.group(2)
        name_start = len(m.group(1))
        name_end = name_start + len(name)

                    uri=state.uri,
                    range=types.Range(
                        start=types.Position(line=def_line, character=0),
                        end=types.Position(line=def_line, character=len(lines[def_line])),
                    ),
                )

    return None


def _find_component_def_line(lines: list, name: str) -> Optional[int]:
    pattern = rf"^\s*{re.escape(name)}\s*=\s*\("
    for i, line in enumerate(lines):
        if re.match(pattern, line):
            return i
    return None

import re
from typing import Optional

from lsprotocol import types

from ..parser import HTML_TAGS
from .state import DocumentState


def get_hover(
    state: DocumentState, line: int, character: int,
) -> Optional[types.Hover]:
    lines = state.source.split("\n")
    if line >= len(lines):
        return None

    current_line = lines[line]
    word = _word_at(current_line, character)
    if not word:
        return None

    # Component def: name = (params):
    m = re.match(r"(\s*)(\w+)\s*=\s*\(([^)]*)\)", current_line)
    if m and m.group(2) == word:
        name = m.group(2)
        params = m.group(3)
        return types.Hover(
            contents=types.MarkupContent(
                kind=types.MarkupKind.Markdown,
                value=f"**component** `{name}({params})`",
            ),
        )

    # Component call
    if word in state.components:
        comp = state.components[word]
        params = ", ".join(comp.params)
        def_line = _find_def_line(lines, word)
        loc = f" (line {def_line + 1})" if def_line is not None else ""
        return types.Hover(
            contents=types.MarkupContent(
                kind=types.MarkupKind.Markdown,
                value=f"**component** `{word}({params})`{loc}",
            ),
        )

    # HTML tag
    if word in HTML_TAGS:
        return types.Hover(
            contents=types.MarkupContent(
                kind=types.MarkupKind.Markdown,
                value=f"**HTML element** `<{word}>`",
            ),
        )

    return None


def _word_at(line: str, col: int) -> Optional[str]:
    if col >= len(line):
        return None
    if not line[col:col+1].isalnum() and line[col:col+1] != "_":
        return None
    start = col
    while start > 0 and (line[start - 1].isalnum() or line[start - 1] == "_"):
        start -= 1
    end = col
    while end < len(line) and (line[end].isalnum() or line[end] == "_"):
        end += 1
    return line[start:end] if start < end else None


def _find_def_line(lines: list, name: str) -> Optional[int]:
    pattern = rf"^\s*{re.escape(name)}\s*=\s*\("
    for i, line in enumerate(lines):
        if re.match(pattern, line):
            return i
    return None
```

def get_document_symbols(state: DocumentState) -> List[types.DocumentSymbol]:
    symbols: List[types.DocumentSymbol] = []
    lines = state.source.split("\n")

    for node in state.ast:
        sym = _node_to_symbol(node, lines)
        if sym is not None:
            symbols.append(sym)

def _node_to_symbol(
    node: Node, lines: List[str],
) -> Optional[types.DocumentSymbol]:
    if isinstance(node, ComponentDef):
        line_num = _find_line(lines, f"{node.name}")
        params = ", ".join(node.params)
        return types.DocumentSymbol(
            name=node.name,
            detail=f"({params})",
            kind=types.SymbolKind.Function,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
        )

    if isinstance(node, StyleElement):
        line_num = _find_line(lines, "style")
        return types.DocumentSymbol(
            name="style",
            kind=types.SymbolKind.Module,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
        )

    if isinstance(node, ScriptElement):
        line_num = _find_line(lines, "script")
        children: List[types.DocumentSymbol] = []
        for sn in node.body_nodes:
            child = _script_node_to_symbol(sn, lines)
            if child:
                children.append(child)
        return types.DocumentSymbol(
            name="script",
            kind=types.SymbolKind.Module,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
            children=children if children else None,
        )

    if isinstance(node, Element):
        line_num = _find_line(lines, node.tag)
        children = []
        for child in node.children:
            sym = _node_to_symbol(child, lines)
            if sym:
                children.append(sym)
        return types.DocumentSymbol(
            name=node.tag,
            kind=types.SymbolKind.Class,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
            children=children if children else None,
        )

    return None


def _script_node_to_symbol(node, lines: List[str]) -> Optional[types.DocumentSymbol]:
    if isinstance(node, ScriptFunction):
        if not node.name:
            return None
        line_num = _find_line(lines, node.name)
        params = ", ".join(node.params)
        return types.DocumentSymbol(
            name=node.name,
            detail=f"({params})",
            kind=types.SymbolKind.Function if not node.is_async else types.SymbolKind.Function,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
        )

    if isinstance(node, ScriptClass):
        line_num = _find_line(lines, f"class {node.name}")
        children = []
        for method in node.methods:
            child = _script_node_to_symbol(method, lines)
            if child:
                children.append(child)
        return types.DocumentSymbol(
            name=node.name,
            detail=f"extends {node.extends}" if node.extends else None,
            kind=types.SymbolKind.Class,
            range=_line_range(line_num, lines),
            selection_range=_line_range(line_num, lines),
            children=children if children else None,
        )

    return None


def _find_line(lines: List[str], needle: str) -> int:
    for i, line in enumerate(lines):
        if needle in line:
            return i
    return 0


def _line_range(line_num: int, lines: List[str]) -> types.Range:
    length = len(lines[line_num]) if line_num < len(lines) else 0
    return types.Range(
        start=types.Position(line=line_num, character=0),
        end=types.Position(line=line_num, character=length),
    )
```
